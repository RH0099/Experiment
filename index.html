function autoPrepareReports(){
  document.getElementById('abuseMail').value=
`Subject: Abuse Report (${CATEGORY})\n\nReported URL: ${LAST_URL}\nRisk Level: ${RISK}\nCategory: ${CATEGORY}\n\nThis website is suspected to be harmful and violates acceptable use policies.`;

  document.getElementById('publicWarn').value=
`⚠️ PUBLIC WARNING ⚠️\n\nThe website ${LAST_URL} has been identified as ${CATEGORY} with ${RISK} risk.\nUsers are advised to avoid interaction.`;
}

function reportGSB(){
  window.open('https://safebrowsing.google.com/safebrowsing/report_phish/?url='+encodeURIComponent(LAST_URL),'_blank');
}
function reportVT(){
  window.open('https://www.virustotal.com/gui/search/'+encodeURIComponent(LAST_URL),'_blank');
}
function sendMail(){
  window.location.href='mailto:abuse@provider.com?body='+encodeURIComponent(document.getElementById('abuseMail').value);
}
function reportRegistrar(){
  window.open('https://www.icann.org/compliance/complaint','_blank');
}
function showWarning(){
  alert(document.getElementById('publicWarn').value);
